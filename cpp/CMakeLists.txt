cmake_minimum_required(VERSION 3.20)
project(trychert VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(CURL 7.84 REQUIRED)
find_package(nlohmann_json 3.9 REQUIRED CONFIG)

add_library(trychert src/client.cpp src/facetime.cpp)
add_library(trychert::trychert ALIAS trychert)
# Source-build compatibility for the unpublished predecessor.
add_library(cherthq::imessage ALIAS trychert)
set_target_properties(trychert PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
target_compile_features(trychert PUBLIC cxx_std_20)
target_include_directories(trychert PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(trychert PUBLIC nlohmann_json::nlohmann_json PRIVATE CURL::libcurl)
if(MSVC)
  target_compile_options(trychert PRIVATE /W4 /WX)
else()
  target_compile_options(trychert PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(CHERT_SANITIZERS "Enable address and undefined behavior checks" OFF)
if(CHERT_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "This sanitizer configuration requires GCC or Clang")
  endif()
  target_compile_options(trychert PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(trychert PUBLIC -fsanitize=address,undefined)
endif()

install(TARGETS trychert EXPORT trychertTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/trychert)
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/trychert/examples)
configure_package_config_file(cmake/trychertConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/trychertConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/trychert)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/trychertConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(EXPORT trychertTargets NAMESPACE trychert:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/trychert)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/trychertConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/trychertConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/trychert)

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(fixture IN ITEMS client route_fixture pipeline_fixture text_send typed_text_fixture reaction_send typed_reaction_fixture unified server_contract)
    if(fixture STREQUAL "client")
      set(executable chert-sdk-test)
    elseif(fixture STREQUAL "text_send")
      set(executable chert-sdk-text-send-test)
    elseif(fixture STREQUAL "reaction_send")
      set(executable chert-sdk-reaction-send-test)
    else()
      string(REPLACE "_" "-" name ${fixture})
      set(executable chert-sdk-${name})
    endif()
    add_executable(${executable} tests/${fixture}.cpp)
    target_link_libraries(${executable} PRIVATE trychert::trychert)
    if(MSVC)
      target_compile_options(${executable} PRIVATE /W4 /WX)
    else()
      target_compile_options(${executable} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endforeach()
  add_test(NAME client-http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_fixture.py $<TARGET_FILE:chert-sdk-test>)
  add_test(NAME text-send-http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/text_send_http_fixture.py $<TARGET_FILE:chert-sdk-text-send-test>)
  add_test(NAME reaction-send-http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reaction_send_http_fixture.py $<TARGET_FILE:chert-sdk-reaction-send-test>)
  add_test(NAME unified-http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/unified_http_fixture.py $<TARGET_FILE:chert-sdk-unified>)
  add_test(NAME clean-consumer COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/clean_consumer.py ${CMAKE_COMMAND} ${CMAKE_CURRENT_BINARY_DIR})
  set(CHERT_FACETIME_SOURCE "" CACHE PATH "Optional cherthq/facetime control-plane checkout with installed dependencies")
  if(CHERT_FACETIME_SOURCE)
    find_program(BUN_EXECUTABLE bun REQUIRED)
    add_test(NAME facetime-server-contract COMMAND ${BUN_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/server_contract.mjs
      ${CHERT_FACETIME_SOURCE} $<TARGET_FILE:chert-sdk-server-contract>)
  endif()
endif()

set(CPACK_PACKAGE_NAME chert-sdk)
set(CPACK_SOURCE_PACKAGE_FILE_NAME "chert-sdk-${PROJECT_VERSION}-Source")
set(CPACK_PACKAGE_VENDOR "Chert Technologies")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Chert iMessage and FaceTime C++20 HTTP SDK")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/cherthq/sdk")
if(BUILD_SHARED_LIBS)
  set(CPACK_PACKAGE_FILE_NAME "chert-sdk-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}-shared")
else()
  set(CPACK_PACKAGE_FILE_NAME "chert-sdk-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}-static")
endif()
set(CPACK_RESOURCE_FILE_LICENSE ${CMAKE_CURRENT_SOURCE_DIR}/LICENSE)
set(CPACK_GENERATOR TGZ)
set(CPACK_SOURCE_GENERATOR TGZ)
# Match source children, not an ancestor that happens to be a scratch directory.
set(CPACK_SOURCE_IGNORE_FILES "${CMAKE_CURRENT_SOURCE_DIR}/build[^/]*/;${CMAKE_CURRENT_SOURCE_DIR}/\\.git/;${CMAKE_CURRENT_SOURCE_DIR}/\\.amp/;__pycache__;\\.pyc$;\\.DS_Store$")
include(CPack)
