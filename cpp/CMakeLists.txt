cmake_minimum_required(VERSION 3.20)
project(ChertSDK VERSION 0.1.0 LANGUAGES CXX)
find_package(CURL 7.84 REQUIRED)
find_path(CHERT_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(cherthq-imessage src/client.cpp)
add_library(cherthq::imessage ALIAS cherthq-imessage)
target_compile_features(cherthq-imessage PUBLIC cxx_std_20)
target_include_directories(cherthq-imessage PUBLIC include ${CHERT_JSON_INCLUDE_DIR})
target_link_libraries(cherthq-imessage PRIVATE CURL::libcurl)
if(MSVC)
  target_compile_options(cherthq-imessage PRIVATE /W4 /WX)
else()
  target_compile_options(cherthq-imessage PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(CHERT_SANITIZERS "Enable address and undefined behavior checks" OFF)
if(CHERT_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "This sanitizer configuration requires GCC or Clang")
  endif()
  target_compile_options(cherthq-imessage PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cherthq-imessage PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(chert-sdk-test tests/client.cpp)
  target_link_libraries(chert-sdk-test PRIVATE cherthq::imessage)
  add_executable(chert-sdk-route-fixture tests/route_fixture.cpp)
  target_link_libraries(chert-sdk-route-fixture PRIVATE cherthq::imessage)
  add_executable(chert-sdk-pipeline-fixture tests/pipeline_fixture.cpp)
  target_link_libraries(chert-sdk-pipeline-fixture PRIVATE cherthq::imessage)
  if(MSVC)
    target_compile_options(chert-sdk-test PRIVATE /W4 /WX)
    target_compile_options(chert-sdk-route-fixture PRIVATE /W4 /WX)
    target_compile_options(chert-sdk-pipeline-fixture PRIVATE /W4 /WX)
  else()
    target_compile_options(chert-sdk-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(chert-sdk-route-fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(chert-sdk-pipeline-fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME client-http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_fixture.py $<TARGET_FILE:chert-sdk-test>)
endif()
